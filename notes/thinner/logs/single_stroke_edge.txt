// SCRATCH (untracked; deleted after): does a single thinned stroke's free
// edge show a rim the slow flow never levels? One stroke on a plain ground.
use paint::color::linear_to_srgb;
use paint::{Canvas, Gesture, Held, Palette, Tool, hex};
#[test]
#[ignore]
fn single_stroke_edge() {
    let pal = Palette::named_box("inness").unwrap();
    let idx = |n: &str| pal.tubes.iter().position(|t| t.name == n).unwrap();
    let mix = pal.pile(vec![(idx("raw umber"), 2.0 / 3.0), (idx("bone black"), 1.0 / 3.0)]);
    let out = std::env::var("DIAG_OUT").unwrap();
    for (name, t, wait) in [("thinned_0min", 0.5f32, 0.0f32), ("thinned_30min", 0.5, 30.0), ("unthinned", 0.0, 0.0)] {
        let mut c = Canvas::new(256, 1.5, hex("#e8dcc0")).with_engine(3).with_size_mm(300.0);
        c.prime(hex("#e3d6b8"), 0.85, 40.0, 0.5, 0.0, 2);
        let mut h = Held::new(Tool::filbert(60.0), 5);
        let p = mix.laid(0.0);
        h.load(if t > 0.0 { p.with_thinner(t) } else { p }, 0.6);
        c.drag(&mut h, &Gesture::new(vec![(150.0, 330.0), (850.0, 340.0)]).pressure(0.8, 0.8), None);
        if wait > 0.0 {
            c.wait(wait);
        }
        let f = c.frame();
        // paint across the stroke at x = 500, every pixel row
        let prof: Vec<String> = (0..f.h).map(|y| f.uy(y)).filter(|&y| (270.0..400.0).contains(&y)).map(|y| format!("{:.1}", (480..=520).map(|x| c.wet_um(x as f32, y)).sum::<f32>() / 41.0)).collect();
        println!("{name}: paint µm across the stroke (y 270..400, one value a pixel row): {}", prof.join(" "));
        let px: Vec<u8> = c.seen().iter().flat_map(|p| p.map(|v| (linear_to_srgb(v).clamp(0.0, 1.0) * 255.0).round() as u8)).collect();
        image::RgbImage::from_raw(f.w as u32, f.h as u32, px).unwrap().save(format!("{out}/{name}.png")).unwrap();
    }
}
